{{- /*
  .github/workflows/dependency-review.yml. The host file uses #[[[ ]]]
  delimiters, but partials are always parsed with {{ }}, so GitHub Actions
  expressions are emitted as raw-string literals.

  Knobs: .Computed.depSubmission (optional). "mix" adds the
  erlef/mix-dependency-submission step, the egress it needs, and contents:
  write.
*/ -}}
{{- $mix := eq (default "" .Computed.depSubmission) "mix" -}}
# Dependency Review Action
#
# This Action will scan dependency manifest files that change as part of a Pull Request,
# surfacing known-vulnerable versions of the packages declared or updated in the PR. Once
# installed, if the workflow run is marked as required, PRs introducing known-vulnerable
# packages will be blocked from merging.
#
# Source repository: https://github.com/actions/dependency-review-action
name: 'Dependency Review'
on:
  push:
    branches:
      - main
  pull_request:

permissions: {}

concurrency:
  group: {{ `${{ github.workflow }}-${{ github.event.pull_request.number || github.ref }}` }}
  cancel-in-progress: true

jobs:
  dependency-review:
    name: Review Dependencies
    if: {{ `${{ !github.event.repository.private }}` }}
    permissions:
{{- if $mix }}
      contents: write # Required to submit dependencies
{{- else }}
      contents: read
{{- end }}
      pull-requests: write # Required to write comments on the PR

    runs-on: ubuntu-latest
    steps:
      - name: Harden the runner
        if: {{ `${{ !github.event.repository.private }}` }}
        uses: step-security/harden-runner@aebdcf6156ab6691adf63218240e476564a1ed4c # v2.23.0
        with:
          disable-sudo: true
          egress-policy: block
          allowed-endpoints: >
            api.deps.dev:443
            api.github.com:443
            api.securityscorecards.dev:443
            github.com:443
{{- if $mix }}
            hex.pm:443
            release-assets.githubusercontent.com:443
            tmaproduction.blob.core.windows.net:443
            tuf-repo-cdn.sigstore.dev:443
            tuf-repo.github.com:443
{{- end }}

      - uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1 # v7.0.1
        with:
          persist-credentials: false
{{- if $mix }}

      - name: Submit dependencies
        uses: erlef/mix-dependency-submission@c5bd30a0e18822272a914adb7fb153866d491baf # v1.3.5
{{- end }}

      - name: Dependency Review
        if: {{ `${{ github.event_name == 'pull_request' }}` }}
        uses: actions/dependency-review-action@bbc44d14a204704ccc493733616cd1e93a0321ec # v5.0.1
        with:
          fail-on-severity: low
          license-check: true
          vulnerability-check: true
          comment-summary-in-pr: on-failure
